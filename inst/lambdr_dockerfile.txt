FROM public.ecr.aws/lambda/provided:al2023

ENV R_VERSION=4.6.1

# Posit's R build for RHEL 9; Amazon Linux 2023 provides all its dependencies.
# microdnf can't install an RPM from a file, so install the dependencies
# first and then the RPM with rpm.
RUN curl -fsSLO https://cdn.posit.co/r/rhel-9/pkgs/R-${R_VERSION}-1-1.x86_64.rpm \
  && dnf -y --setopt=install_weak_deps=0 install \
    $(rpm -qpR R-${R_VERSION}-1-1.x86_64.rpm | grep -v '^rpmlib' | awk '{print $1}') \
    openssl-devel \
  && rpm -i R-${R_VERSION}-1-1.x86_64.rpm \
  && rm R-${R_VERSION}-1-1.x86_64.rpm \
  && dnf clean all

ENV PATH="${PATH}:/opt/R/${R_VERSION}/bin/"
# Install binary packages, also when restoring an renv.lock.
ENV RENV_CONFIG_REPOS_OVERRIDE="https://packagemanager.posit.co/cran/__linux__/rhel9/latest"

RUN Rscript -e 'install.packages(c("lambdr", "renv"), repos = "https://packagemanager.posit.co/cran/__linux__/rhel9/latest")'

RUN printf '#!/bin/sh\ncd /lambda\nRscript runtime.R' > /var/runtime/bootstrap \
  && chmod +x /var/runtime/bootstrap

WORKDIR /lambda
COPY runtime.R runtime.R
RUN chmod 755 -R /lambda
